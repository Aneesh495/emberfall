cmake_minimum_required(VERSION 3.25)
project(Emberfall VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
option(EMBERFALL_FRONTEND "Build the native SDL3/Vulkan programs" ON)
option(EMBERFALL_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)

add_library(emberfall_options INTERFACE)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(emberfall_options INTERFACE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    if(EMBERFALL_SANITIZE)
        target_compile_options(emberfall_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(emberfall_options INTERFACE -fsanitize=address,undefined)
    endif()
endif()

add_library(emberfall_core src/math.cpp src/scene.cpp src/files.cpp src/collision.cpp src/game.cpp src/campaign.cpp src/editor.cpp src/net_protocol.cpp src/net_session.cpp src/scheduler.cpp src/replay.cpp src/dynamics.cpp src/scene_stream.cpp)
target_include_directories(emberfall_core PUBLIC include)
target_link_libraries(emberfall_core PUBLIC emberfall_options)
if(BUILD_TESTING)
    add_executable(core_tests tests/core.cpp)
    target_link_libraries(core_tests PRIVATE emberfall_core)
    add_test(NAME core COMMAND core_tests)
endif()
if(BUILD_TESTING)
    add_executable(world_tests tests/world.cpp)
    target_link_libraries(world_tests PRIVATE emberfall_core)
    add_test(NAME world COMMAND world_tests)
endif()
if(BUILD_TESTING)
    add_executable(scene_tests tests/scene.cpp)
    target_link_libraries(scene_tests PRIVATE emberfall_core)
    add_test(NAME scene COMMAND scene_tests)
endif()
if(BUILD_TESTING)
    add_executable(collision_tests tests/collision.cpp)
    target_link_libraries(collision_tests PRIVATE emberfall_core)
    add_test(NAME collision COMMAND collision_tests)
endif()


if(BUILD_TESTING)
    add_executable(playthrough_tests tests/playthrough.cpp)
    target_link_libraries(playthrough_tests PRIVATE emberfall_core)
    add_test(NAME playthrough COMMAND playthrough_tests "${PROJECT_SOURCE_DIR}/assets/levels")
endif()
if(BUILD_TESTING)
    add_executable(campaign_tests tests/campaign.cpp)
    target_link_libraries(campaign_tests PRIVATE emberfall_core)
    add_test(NAME campaign COMMAND campaign_tests "${PROJECT_SOURCE_DIR}/assets/levels")
endif()
if(BUILD_TESTING)
    add_executable(editor_tests tests/editor.cpp)
    target_link_libraries(editor_tests PRIVATE emberfall_core)
    add_test(NAME editor_history COMMAND editor_tests "${PROJECT_SOURCE_DIR}/assets/levels/01-aperture.scene")
endif()
if(BUILD_TESTING)
    add_executable(network_tests tests/network.cpp)
    target_link_libraries(network_tests PRIVATE emberfall_core)
    add_test(NAME network COMMAND network_tests "${PROJECT_SOURCE_DIR}/assets/levels")
endif()
find_package(Threads REQUIRED)
target_link_libraries(emberfall_core PUBLIC Threads::Threads)
if(BUILD_TESTING)
    add_executable(scheduler_tests tests/scheduler.cpp)
    target_link_libraries(scheduler_tests PRIVATE emberfall_core)
    add_test(NAME scheduler COMMAND scheduler_tests)
    set_tests_properties(scheduler PROPERTIES TIMEOUT 25)
endif()

add_executable(ember-tool src/tools/main.cpp)
target_link_libraries(ember-tool PRIVATE emberfall_core)

find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
add_custom_target(emberfall_levels
    COMMAND "${Python3_EXECUTABLE}" "${PROJECT_SOURCE_DIR}/scripts/cook.py"
        "${PROJECT_SOURCE_DIR}/assets/levels" "${CMAKE_CURRENT_BINARY_DIR}/levels"
        "$<TARGET_FILE:ember-tool>"
    DEPENDS ember-tool VERBATIM)

if(EMBERFALL_FRONTEND)
    include(cmake/Frontend.cmake)
endif()
if(BUILD_TESTING)
    add_test(NAME cooker COMMAND "${Python3_EXECUTABLE}" "${PROJECT_SOURCE_DIR}/tests/cooker.py"
        "${PROJECT_SOURCE_DIR}/scripts/cook.py" "${PROJECT_SOURCE_DIR}/assets/levels" "$<TARGET_FILE:ember-tool>")
endif()
if(BUILD_TESTING)
    add_executable(dynamics_tests tests/dynamics.cpp)
    target_link_libraries(dynamics_tests PRIVATE emberfall_core)
    add_test(NAME dynamics COMMAND dynamics_tests)
endif()
if(BUILD_TESTING)
    add_executable(runtime_bench tests/bench_runtime.cpp)
    target_link_libraries(runtime_bench PRIVATE emberfall_core)
endif()
if(BUILD_TESTING)
    add_executable(stream_tests tests/stream.cpp)
    target_link_libraries(stream_tests PRIVATE emberfall_core)
    add_test(NAME scene_stream COMMAND stream_tests "${PROJECT_SOURCE_DIR}/assets/levels/01-aperture.scene")
endif()
