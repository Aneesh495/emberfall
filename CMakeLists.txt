cmake_minimum_required(VERSION 3.25)
project(Emberfall VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
option(EMBERFALL_FRONTEND "Build the native SDL3/Vulkan programs" ON)
option(EMBERFALL_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)

add_library(emberfall_options INTERFACE)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(emberfall_options INTERFACE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    if(EMBERFALL_SANITIZE)
        target_compile_options(emberfall_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(emberfall_options INTERFACE -fsanitize=address,undefined)
    endif()
endif()

add_library(emberfall_core src/math.cpp)
target_include_directories(emberfall_core PUBLIC include)
target_link_libraries(emberfall_core PUBLIC emberfall_options)
if(BUILD_TESTING)
    add_executable(core_tests tests/core.cpp)
    target_link_libraries(core_tests PRIVATE emberfall_core)
    add_test(NAME core COMMAND core_tests)
endif()
